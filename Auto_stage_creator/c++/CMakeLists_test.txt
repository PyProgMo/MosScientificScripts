# Simple CMake file for testing core functionality without Qt
cmake_minimum_required(VERSION 3.16)
project(AutoStageTest)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Build test executable (no Qt dependencies)
add_executable(test_coord
    test.cpp
    AutoStageCoordCreator.cpp
)

# Set output directory
set_target_properties(test_coord PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
)
